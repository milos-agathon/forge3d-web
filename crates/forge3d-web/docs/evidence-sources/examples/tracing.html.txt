  <h1>Progressive tracing</h1>
  <p>A sphere and torus, joined with smooth CSG, above a triangle floor.</p>
  <button id="render">Render 64 samples</button
  ><button id="cancel">Cancel</button
  ><canvas id="image" width="320" height="200"></canvas
  ><output id="status">Ready</output>
  <script type="module">
    const sdk = await import(
      /* @vite-ignore */ new URLSearchParams(location.search).has("dist")
        ? "/dist/index.js"
        : "/src-ts/index.ts"
    );
    window.__w17 = sdk;
    let controller, active;
    document.querySelector("#cancel").onclick = () => controller?.abort();
    document.querySelector("#render").onclick = async () => {
      if (active) return;
      document.querySelector("#render").disabled = true;
      controller = new AbortController();
      const status = document.querySelector("#status"),
        canvas = document.querySelector("canvas");
      try {
        active = await sdk.PathTracer.create({
          width: 320,
          height: 200,
          seed: 1597463007,
          maxBounces: 4,
          engine: "wavefront",
          camera: { origin: [3, 2, 5], target: [0, 0, 0] },
        });
        const builder = new sdk.SdfSceneBuilder(),
          sphere = builder.addSphere([0, 0, 0], 0.8, 0),
          torus = builder.addTorus([0, 0.3, 0], 1, 0.18, 1);
        builder.smoothUnion(sphere, torus, 0.12);
        await active.setScene({
          sdf: builder.build(),
          materials: [
            { albedo: [0.7, 0.3, 0.12] },
            { albedo: [0.15, 0.5, 0.7], metallic: 0.6, roughness: 0.25 },
            { albedo: [0.5, 0.5, 0.5] },
          ],
          triangles: [
            {
              v0: [-20, -1, -20],
              v1: [20, -1, 20],
              v2: [20, -1, -20],
              materialId: 2,
              id: 1,
            },
            {
              v0: [-20, -1, -20],
              v1: [-20, -1, 20],
              v2: [20, -1, 20],
              materialId: 2,
              id: 2,
            },
          ],
          lights: [{ position: [-2, 4, 3], intensity: 50 }],
          environment: [0.4, 0.5, 0.7],
        });
        for await (const update of active.progressive({
          spp: 64,
          updateEvery: 4,
          signal: controller.signal,
        })) {
          const bytes = new Uint8ClampedArray(320 * 200 * 4);
          for (let p = 0; p < 320 * 200; p++) {
            for (let c = 0; c < 3; c++) {
              const v = update.result.color[p * 3 + c],
                x = v / (1 + v);
              bytes[p * 4 + c] =
                255 *
                (x <= 0.0031308
                  ? 12.92 * x
                  : 1.055 * Math.pow(x, 1 / 2.4) - 0.055);
            }
            bytes[p * 4 + 3] = 255;
          }
          canvas
            .getContext("2d")
            .putImageData(new ImageData(bytes, 320, 200), 0, 0);
          status.value = update.samples + " / 64 samples";
        }
      } catch (error) {
        status.value = error.code + ": " + error.message;
      } finally {
        active?.dispose();
        active = undefined;
        document.querySelector("#render").disabled = false;
      }
    };
  </script>
